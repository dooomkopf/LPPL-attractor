% ─── 2. Laplace Diffusion ─────────────────────────────────────────────────────
\noindent\textbf{\color{cdiff}2.\;Diffusion Extension}

\begin{align}
  \dot{y}_2 &= -\frac{\kappa\,\operatorname{sgn}[X(t)] + y_2}{t}
               + \frac{\sigma(t)}{t}\,\dot{W}_t \\[4pt]
  X(t)      &= n(t) - \mu(t), \qquad n(t) := n_{PL} + t\,y_2
\end{align}

\smallskip
\noindent{\color{muted}\small For the sign-diffusion part alone and a fixed target $\mu$, the exponent
$n = n_{PL} + t\,y_2$ obeys $dn = -\kappa\,\operatorname{sgn}[X]\,dt + \sigma\,dW_t$, whose stationary
distribution is Laplace$(\mu, b)$ with $b = \sigma^2/(2\kappa)$. The noise is therefore set by the
measured Laplace scale $b_c$ of the daily exponent per cycle, $\sigma = \sqrt{2\kappa b_c}$. The full
system, with spring, $z$ and a time-dependent $\mu$, has no stationary Laplace law.}

\smallskip
\noindent{\color{cdiff}\textbf{Target exponent $\mu(t)$.}}
{\color{fg}\small In the simplest case $\mu(t)$ is piecewise constant per halving cycle $c$ and phase,
\[
  \mu(t) = n_{PL} + \Delta_{c,P},\qquad P\in\{P_1,P_2,P_3\},
\]
with $P_1\!: 100 \le d_c \le 550$, $P_2\!: 550 < d_c \le 950$ and $P_3$ otherwise, where $d_c$ is the
position within the halving cycle, rescaled from the actual cycle length to a reference length of
$1392.5$\,d. The offsets $\Delta_{c,P}$ are the measured mean deviations of the daily exponent from
$n_{PL}$. At phase boundaries $\mu$ jumps, so $X$ jumps while $n$ evolves continuously.
Alternatively, $\mu(t)$ can follow a centred moving average of the measured exponent
(window 30--90\,d).}
